Public Function HightlightDups(co1 As Range, co2 As Range) As Range

Dim highlights As Range
Dim cell As Range
Dim cell2 As Range



For each cell In co1

'Highlighting in User Name
	If WorksheetFunction.CountIf(co2, cell.Value) > 0 Then

		cell.Interior.Color = RGB(255, 255, 153)
	
		If highlights Is Nothing Then 
			Set highlights = cell.EntireRow
		Else
			Set highlights = Union(highlights, cell.EntireRow)
		End If
	End If

	Next cell

'Highlighting in E3 coloumn

	For each cell2 In co2

	If WorksheetFunction.CountIf(co1, cell2.Value) > 0 Then

		cell2.Interior.Color = RGB(255, 255, 153)
	
	End If

	Next cell2


'Return
	Set HightlightDups = highlights

End Function
