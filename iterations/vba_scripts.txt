Sub AddFilenametoColumn()


Dim result() As String
Dim ws As Worksheet
Dim targetRow As Long
Dim lastCol As Long
Dim nextCol As Long
Dim valueToFilename As Variant
Dim Cell As Range

Set ws = ThisWorkbook.Sheets("v6")

For Each Cell In Selection

    result = Split(Cell.Value, vbLf)
    
    targetRow = Cell.Row
    
    If InStr(1, result(0), "handle scavenged", vbTextCompare) > 0 Then
        valueToFilename = result(3)
    
    Else
        valueToFilename = result(7)
    
    End If
    
    
    nextCol = ws.Cells(targetRow, ws.Columns.Count).End(xlToLeft).Column + 1
    
    ws.Cells(targetRow, 7).Value = valueToFilename
    
    Next Cell

End Sub
