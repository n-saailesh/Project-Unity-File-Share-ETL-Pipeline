Sub insertingValues()

Dim Cell As Range
Dim targetRow As Long

Set ws = ThisWorkbook.Sheets("Data 2026-07-15 (13 40)")

For Each Cell In Selection

    securityID = getSecurityID(Cell.Row)
    accountName = getAccountName(Cell.Row)
    accountDomain = getAccountDomain(Cell.Row)
    shareName = getShareName(Cell.Row)
    relativeTargetName = getRelativeTargetName(Cell.Row)
    sourceAddress = getSourceAddress(Cell.Row)
    sharePath = getSharePath(Cell.Row)
    sharePathName = getSharePathName(Cell.Row)
    
    targetRow = Cell.Row
    
    ws.Cells(targetRow, 8).Value = securityID
    ws.Cells(targetRow, 9).Value = accountName
    ws.Cells(targetRow, 10).Value = accountDomain
    ws.Cells(targetRow, 11).Value = shareName
    ws.Cells(targetRow, 12).Value = sharePath
    ws.Cells(targetRow, 13).Value = sharePathName
    ws.Cells(targetRow, 14).Value = relativeTargetName
    ws.Cells(targetRow, 15).Value = sourceAddress
    
Next Cell

End Sub
